```sql
-- dept, staff
create view v_staff
as
select s.id, s.name, d.name dept, d.city, s.salary
from staff s
left join dept d
on s.dept_id=d.id
order by salary desc;

-- 各个部门的人数
create view v_dept 
as
select d.id, d.name, count(s.id) n
from dept d
left join staff s
on d.id=s.dept_id
group by d.id, d.name
order by d.id;

select * from v_dept;

create view v_staff_male
as
select id,name,gender
from staff
where gender='male';

```

视图里的数据能否更新和删除？
- 数组来自一张表且没有使用函数，都是原始数据；可以插入、更新和删除
- 数据使用函数、来多张表，不可以进行更新、插入和删除

```sql
DROP TABLE IF EXISTS `user` ;

CREATE TABLE IF NOT EXISTS `user` (
  `id` INT NOT NULL AUTO_INCREMENT,
  `name` VARCHAR(45) NOT NULL,
  `about` VARCHAR(45) NULL,
  PRIMARY KEY (`id`),
  UNIQUE INDEX `name_UNIQUE` (`name` ASC) VISIBLE)
ENGINE = InnoDB;


-- -----------------------------------------------------
-- Table `mydb`.`login`
-- -----------------------------------------------------
DROP TABLE IF EXISTS `login` ;

CREATE TABLE IF NOT EXISTS `login` (
  `id` INT NOT NULL,
  `name` VARCHAR(45) NOT NULL,
  `passwd` CHAR(64) NOT NULL,
  PRIMARY KEY (`id`),
  CONSTRAINT `login_fk_user`
    FOREIGN KEY (`id`)
    REFERENCES `mydb`.`user` (`id`)
    ON DELETE CASCADE
    ON UPDATE CASCADE)
ENGINE = InnoDB;
```

```sql
CREATE PROCEDURE `reg`(
    name varchar(16), 
    passwd varchar(8)
)
BEGIN
	insert into user(`name`) values(name);
    insert into login values(last_insert_id(), name, sha2(passwd, 256));
    commit;
END

call reg('bob', '123');

## MySQL C API

### 核心数据（结构体）
- MYSQL         数据库连接
- MYSQL_RES     执行 SQL 获得资源，结果集
- MYSQL_ROW     从结果集获得的一行数据，记录，字符串数组
- MYSQL_FIELD   列、字段的元数据：列名、类型、大小

// my.c
// MySQL C API
#include <stdio.h>
#include <stdlib.h>

#include <mysql/mysql.h>

#define HOST "127.0.0.1"
#define USER "tom"
#define PASSWD "tom"
#define DB "hr"
#define PORT 3306

int main(int argc, char const *argv[])
{
    // 初始化数据库连接
    MYSQL *conn = mysql_init(NULL);
    if (!conn)
    {
        perror("初始化失败\n");
        return EXIT_FAILURE;
    }

    // 建立连接
    if (!mysql_real_connect(conn, HOST, USER, PASSWD, DB, PORT, NULL, 0))
    {
        fprintf(stderr, "连接失败：%d, %s\n", mysql_errno(conn), mysql_error(conn));
        mysql_close(conn);
        return EXIT_FAILURE;
    }

    // 执行 DML
    char *sql = "SELECT id,name,city FROM dept";

    // SQL注入
    // char *sql = "INSERT INTO dept VALUES(NULL,'Test','Changsha')";
    // char *dept = "abc";
    // char *city = "xyz";
    // char sql[128] = {0};
    // sprintf(sql, "INSERT INTO dept VALUES(NULL,'%s','%s')", dept, city);
    // printf("%s\n", sql);

    if (mysql_query(conn, sql))
    {
        fprintf(stderr, "执行查询：%s\n", mysql_error(conn));
        mysql_close(conn);
        return EXIT_FAILURE;
    }
    // 获得结果集
    MYSQL_RES *res = mysql_store_result(conn);
    printf("行：%ld, 列：%d\n", res->row_count, res->field_count);
    
    // 行，记录：字符串数组
    MYSQL_ROW row;
    // int cols = mysql_field_count(conn);
    int cols = res->field_count;

    // 列，字段
    MYSQL_FIELD *field;
    // 遍历字段
    // while ((field = mysql_fetch_field(res)) != NULL)
    // {
    //     printf("%s, %d, %ld\n", field->name, field->type, field->length);
    // }

    // 获得所有字段构成的数组 
    field = mysql_fetch_fields(res);
    for (int i = 0; i < res->field_count; i++)
    {
        printf("%s\t", field[i].name);
    }
    printf("\n");
    
    while ((row = mysql_fetch_row(res)) != NULL)
    {
        // 遍历每一列
        for (int i = 0; i < cols; i++)
        {
            printf("%s\t", row[i]);
        }
        printf("\n");
    }
    mysql_free_result(res);     // 释放结果集

    printf("成功连接\n");
    mysql_close(conn);
    return 0;
}

```
