### Singleton 单例
- 类只能创建一个对象
```c
// singleton.cxx

#include <iostream>
using namespace std;
class Singleton
{
private:
    Singleton()
    {
        cout << "构造函数" << endl;
    } // 1. 构造函数私有化

public:
    ~Singleton() {}
    // 饿汉式：调用之前，提取初始化了资源
    static Singleton instance;

    // 2. 开放一个静态接口
    static Singleton *getInstance()
    {
        // 懒汉式：lazy 延迟加载初始化 第一次调用才创建对象，分配空间
        // static Singleton instance; // 类变量，一份
        return &instance; // 返回的同一份
    }
};
// 类变量的初始化
Singleton Singleton::instance;
int main()
{
    Singleton *s1 = Singleton::getInstance();
    Singleton *s2 = Singleton::getInstance();
    Singleton *s3 = Singleton::getInstance();
    cout << s1 << endl;
    cout << s1 << endl;
    cout << s1 << endl;

    return 0;
}

```

```c
// factory.cxx

#include <iostream>
#include <string>

using namespace std;
class DOC
{
public:
    string name;
    DOC() : name("default") {}
    DOC(string name) : name(name) {}
    ~DOC() {}
    void open()
    {
        cout << "OPEN: " << name << endl;
    }
    virtual void edit() = 0;
};
class Word : public DOC
{
public:
    Word(string name) : DOC(name) {}
    ~Word() {}
    void edit() override
    {
        cout << "EDIT Word " << name << endl;
    }
};

class Ppt : public DOC
{
public:
    Ppt(string name) : DOC(name) {}
    ~Ppt() {}
    void edit() override
    {
        cout << "EDIT ppt " << name << endl;
    }
};

class Excel : public DOC
{
public:
    Excel(string name) : DOC(name) {}
    ~Excel() {}
    void edit() override
    {
        cout << "EDIT Excel " << name << endl;
    }
};

enum DocType
{
    WORD,
    PPT,
    EXCEL,
    PDF
};
class WPS
{
public:
    WPS() {}
    ~WPS() {}

    // 静态工厂
    static DOC *create(DocType type, string name)
    {
        DOC *doc;
        switch (type)
        {
        case WORD:
            doc = new Word(name);
            break;
        case PPT:
            doc = new Ppt(name);
            break;
        case EXCEL:
            doc = new Excel(name);
            break;
        default:
            break;
        }
        return doc;
    }
};

int main()
{
    DOC *d;
    d = WPS::create(WORD, "简历");
    d->open();
    d->edit();
    return 0;
}


// factory2.cxx
// 抽象工厂

#include <iostream>
using namespace std;

// 产品抽象
class Phone
{

public:
    Phone(/* args */) {}
    ~Phone() {}
    virtual void show() = 0;
};
class Pc
{

public:
    Pc(/* args */) {}
    ~Pc() {}
    virtual void show() = 0;
};

class MiPhone : public Phone
{

public:
    MiPhone(/* args */) {}
    ~MiPhone() {}

    void show()
    {
        cout << "MiPhone" << endl;
    }
};

class IPhone : public Phone
{
public:
    IPhone() {}
    ~IPhone() {}
    void show() override
    {
        cout << "IPhone" << endl;
    }
};

class RedMiPc : public Pc
{
public:
    RedMiPc() {}
    ~RedMiPc() {}
    void show() override
    {
        cout << "RedMiPc" << endl;
    }
};

class MacPc : public Pc
{
public:
    MacPc() {}
    ~MacPc() {}
    void show() override
    {
        cout << "MacBook" << endl;
    }
};

class Factory
{

public:
    Factory(/* args */) {}
    ~Factory() {}

    virtual Phone *createPhone() = 0;
    virtual Pc *createPc() = 0;
};

class MiFactory : public Factory
{
public:
    MiFactory() {}
    ~MiFactory() {}

    Phone *createPhone() override
    {
        return new MiPhone();
    }

    Pc *createPc() override
    {
        return new RedMiPc();
    }
};
class AppleFactory : public Factory
{
public:
    AppleFactory() {}
    ~AppleFactory() {}

    Phone *createPhone() override
    {
        return new IPhone();
    }

    Pc *createPc() override
    {
        return new MacPc();
    }
};

// 工厂抽象
int main()
{
    // 使用小米工厂
    Factory *miFactory = new MiFactory();
    Phone *miPhone = miFactory->createPhone();
    Pc *miPc = miFactory->createPc();

    miPhone->show();
    miPc->show();

    delete miPhone;
    delete miPc;
    delete miFactory;

    // 使用苹果工厂
    Factory *appleFactory = new AppleFactory();
    Phone *iPhone = appleFactory->createPhone();
    Pc *macPc = appleFactory->createPc();

    iPhone->show();
    macPc->show();

    delete iPhone;
    delete macPc;
    delete appleFactory;
    return 0;
}
```
