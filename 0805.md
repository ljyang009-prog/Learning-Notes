### 多表查询（表连接）
无效：笛卡尔乘积，行 n*m, 列 n+m
    SELECT * FROM staff,dept;

- 内连接: INNER JOIN（等值连接）
- 外连接
    - 左外连接：LEFT OUTER JOIN
    - 右外连接：RIGHT OUTER JOIN
    - 全连接: FULL OUTER JOIN, UNION

```SQL
-- 内连接
SELECT c1,c1
FROM t1
INNER JOIN t2
ON t1.pk=t2.fk;

SELECT * FROM t1,t2 WHERE t1.pk=t2.fk;

SELECT s.id, s.name, d.name dept, city 
FROM staff s,dept d 
WHERE dept_id=d.id;

SELECT s.id,s.name,d.name dept,city,salary,bonus
FROM staff s
INNER JOIN dept d
ON s.dept_id = d.id;

-- 左连接：join 左侧表中所有记录都出现
SELECT s.id,s.name,d.name dept,city,salary,bonus
FROM staff s
left outer JOIN dept d
ON s.dept_id = d.id;

-- 右连接：join 右侧表中所有记录都出现
SELECT s.id,s.name,d.name dept,city,salary,bonus
FROM staff s
RIGHT outer JOIN dept d
ON s.dept_id = d.id;

-- 全连接(并集 union)
SELECT s.id,s.name,d.name dept,city,salary,bonus
FROM staff s
left outer JOIN dept d
ON s.dept_id = d.id
UNION
SELECT s.id,s.name,d.name dept,city,salary,bonus
FROM staff s
RIGHT outer JOIN dept d
ON s.dept_id = d.id;


CREATE TABLE t1
(
    id int auto_increment primary key,
    val char(4)
);
CREATE TABLE t2
(
    id int auto_increment primary key,
    val char(4)
);
INSERT INTO t1(val) VALUES(1,'A'),(2,'B'),(3,'C');
INSERT INTO t2(val) VALUES(3,'C'),(4,'X'),(5,'Y');

-- 交集：内连接
SELECT * FROM t1,t2 WHERE t1.val=t2.val;
SELECT t1.val FROM t1 INNER JOIN t2 ON t1.val=t2.val;
-- 并集：union 合并
SELECT val FROM t1
UNION
SELECT val FROM t2;

-- union all 不合并相同的行
SELECT val FROM t1
UNION ALL
SELECT val FROM t2;
-- 差集
-- t1 - t2  NOT IN 性能差
SELECT val FROM t1 WHERE val NOT IN (SELECT val FROM t2);
-- t2 - t1
SELECT val FROM t2 WHERE val NOT IN (SELECT val FROM t1);

-- 左连接 + IS NULL
SELECT * FROM t1 LEFT JOIN t2 ON t1.val=t2.val WHERE t2.id IS NULL;
SELECT * FROM t2 LEFT JOIN t1 ON t1.val=t2.val WHERE t1.id IS NULL;

-- `研发`部门的员工信息
SELECT s.*, d.name,d.city FROM staff s 
LEFT JOIN dept d ON s.dept_id=d.id
WHERE d.name='研发';

SELECT * FROM staff
WHERE dept_id=(SELECT id FROM dept WHERE name='研发');
```

### 分组
- GROUP BY
- HAVING

```SQL

    SELECT gender, count(id), max(salary) FROM staff GROUP BY gender;

    SELECT level, count(id), max(salary) FROM staff GROUP BY level;

    SELECT level, count(id), max(salary) max_s
    FROM staff 
    GROUP BY level 
    having max_s > 10000
    order by level desc;

-- 各个部门的名称、人数及平均工资
    SELECT d.name, count(s.id), avg(salary)
    FROM dept d
    LEFT JOIN staff s
    ON d.id=s.dept_id
    GROUP BY d.name;

-- 各个城市、员工人数及最短、最高工资
    SELECT city, count(s.id) n, min(salary), max(salary)
    FROM staff s
    RIGHT JOIN dept d
    ON d.id=s.dept_id
    GROUP BY city
    ORDER BY n DESC
    LIMIT 1;

    SELECT s.id, s.name, b.name boss
    FROM staff s
    LEFT JOIN staff b
    ON s.boss=b.id;

    SELECT b.id, b.name boss, count(s.id) n
    FROM staff s
    right JOIN staff b
    ON s.boss=b.id
    group by b.id, boss
    order by n desc;

    SELECT b.id, b.name boss, count(s.id) N
    FROM staff s
    RIGHT JOIN staff b
    ON s.boss=b.id
    GROUP BY b.id, boss
    order by n DESC;

```

### 模糊查询

使用 like 关键字，配合 _, %

    _ 一个字符
    % 任意个字符

    SELECT name FROM staff WHERE name LIKE 'b%';
    SELECT name FROM staff WHERE name LIKE '%b%';
    SELECT name FROM staff WHERE name LIKE '_a%';
    SELECT name FROM staff WHERE name LIKE '__a%';

### 子查询

在一个查询中嵌套其他查询语句

select * from staff
where salary > (SELECT AVG(salary) FROM staff);

select * from staff
where 
dept_id = (select dept_id from staff where name='tom')
and
birthday < (select birthday from staff where name='tom');
